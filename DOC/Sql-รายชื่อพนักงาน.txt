SELECT
EMP_INTL,EMP_NAME,EMP_SURNME,
EMP_BIRTH,EMP_I_CARD,EMP_TAX_ID,EMP_MARITAL,EMP_GENDER,
FML_ORIGIN,FML_NATNALTY,FML_RELIGION,
PRS_NO,PRS_BANK_AC,PRS_SC_NO,
PRI_START_D,PRI_STATUS,
DEPT_CODE,DEPT_THAIDESC,
TAX_CHDNE_WSP,TAX_CHDNE_NSP,
TAX_CHDWE_WSP,TAX_CHDWE_NSP,
BANK_INITIAL,BR_CODE,BR_THAIDESC,
JBT_THAIDESC
FROM
  EMPFILE,
  FAMILY,
  PERSONALINFO,
  PAYROLLINFO,
  DEPTTAB,
  TAXINFO,
  BANKFILE,
  BRANCH,
  JOBTITLE
WHERE
  (EMPFILE.EMP_KEY =  PERSONALINFO.PRS_EMP)  AND
  (EMPFILE.EMP_KEY =  FAMILY.FML_EMP )  AND
  (EMPFILE.EMP_KEY =  PAYROLLINFO.PRI_EMP)  AND
  (PERSONALINFO.PRS_DEPT  =  DEPTTAB.DEPT_KEY)  AND
  (EMPFILE.EMP_KEY   =  TAXINFO.TAX_EMP)  AND
  (PERSONALINFO.PRS_BANK_C =BANKFILE.BANK_KEY) AND
  (PERSONALINFO.PRS_BR = BRANCH.BR_KEY) AND
  (PERSONALINFO.PRS_JBT = JOBTITLE.JBT_KEY)
